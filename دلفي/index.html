الخطوة 1: التعرف على لغة Delphi وبيئة RAD Studio

Delphi لغة برمجة كائنية التوجه مبنية على Pascal، تشتهر بسرعة تطوير التطبيقات (RAD) وقدرتها على بناء أنظمة سطح مكتب احترافية مرتبطة بقواعد البيانات. بيئتها الرسمية هي RAD Studio.

ما ستتعلمه
  • ما هي Delphi واستخداماتها ومميزاتها.
  • الفرق بينها وبين اللغات الأخرى.
  • التعرف على بيئة RAD Studio ومكوناتها.
  • إنشاء أول مشروع وتشغيله.
أول برنامج
program HelloWorld;

begin
  ShowMessage('Hello Delphi');

الخطوة 2: المتغيرات وأنواع البيانات

المتغير هو مكان في الذاكرة نحفظ فيه قيمة، ولكل متغير نوع يحدد شكل البيانات التي يمكن تخزينها.

الأنواع الأساسية
  • Integer — أعداد صحيحة.
  • Real / Double — أعداد عشرية.
  • String — نصوص.
  • Boolean — صحيح/خطأ.
  • Char — حرف واحد.
  • Currency — مبالغ مالية.
  • TDateTime — التاريخ والوقت.
var
  Name: string;
  Age: Integer;
  Salary: Currency;

begin
  Name := 'Ahmed';
  Age := 25;
  Salary := 5000;

  ShowMessage(Name + ' - ' + IntToStr(Age));
end;
🎯 التدريب: أنشئ متغيرات للاسم والعمر والراتب واعرضها في رسالة واحدة.

الخطوة 3: العمليات الحسابية

نتعلم الجمع والطرح والضرب والقسمة وباقي القسمة والمقارنة. لاحظ أن / تعيد رقماً عشرياً، بينما div تعيد القسمة الصحيحة وmod باقي القسمة.

var
  A, B: Integer;
  Result: Double;

begin
  A := 10;
  B := 3;

  ShowMessage(IntToStr(A + B));      // 13
  ShowMessage(IntToStr(A * B));      // 30
  ShowMessage(IntToStr(A div B));    // 3
  ShowMessage(IntToStr(A mod B));    // 1
  Result := A / B;
  ShowMessage(FloatToStr(Result));   // 3.3333
end;
🎯 المشروع: آلة حاسبة بسيطة بأربع عمليات على نموذج VCL.

الخطوة 4: الجمل الشرطية

تسمح للبرنامج باتخاذ قرار حسب تحقق شرط ما.

الخطوة 5: الحلقات التكرارية

الحلقات تكرر تنفيذ كود معين عدة مرات: for لعدد معلوم، while لشرط مسبق، وrepeat..until لتنفيذ مرة على الأقل.

var
  I: Integer;
  S: string;
begin
  S := '';
  for I := 1 to 10 do
    S := S + IntToStr(I) + ' ';

  ShowMessage(S);

  I := 1;
  while I <= 5 do
  begin
    ShowMessage('While: ' + IntToStr(I));
    Inc(I);
  end;
end;
🎯 المشروع: جدول ضرب الرقم 7 من 1 إلى 10 داخل TMemo.

الخطوة 6: الإجراءات والدوال

Procedure تنفذ عملاً ولا تعيد قيمة، وFunction تعيد قيمة عبر Result. وكلتاهما تقبل معاملات (Parameters).

function Sum(A, B: Integer): Integer;
begin
  Result := A + B;
end;

function IsEven(N: Integer): Boolean;
begin
  Result := (N mod 2 = 0);
end;

procedure ShowWelcome(const Name: string);
begin
  ShowMessage('أهلاً ' + Name);
end;
🎯 المشروع: مكتبة دوال حسابية (جمع، طرح، متوسط، تحقق من الزوجية).

الخطوة 7: المصفوفات Arrays

المصفوفة مجموعة عناصر من نفس النوع تُخزَّن تحت اسم واحد ويُوصل إليها بالفهرس.

var
  Numbers: array[1..5] of Integer;
  I, Total: Integer;
begin
  Numbers[1] := 10;
  Numbers[2] := 20;
  Numbers[3] := 30;
  Numbers[4] := 40;
  Numbers[5] := 50;

  Total := 0;
  for I := 1 to 5 do
    Total := Total + Numbers[I];

  ShowMessage('المجموع = ' + IntToStr(Total));
end;

ويمكن استخدام Dynamic Arrays عند الحاجة لحجم متغير: SetLength(Arr, N);

🎯 المشروع: برنامج لتخزين درجات 10 طلاب وحساب المتوسط وأعلى درجة.

الخطوة 8: التعامل مع النصوص

دوال النصوص من أكثر ما ستستخدمه عملياً عند التعامل مع بيانات المستخدمين والملفات.

var
  S: string;
begin
  S := '  Hello Delphi World  ';

  ShowMessage(IntToStr(Length(S)));         // الطول
  ShowMessage(Trim(S));                     // حذف المسافات
  ShowMessage(UpperCase(Trim(S)));          // تحويل لأحرف كبيرة
  ShowMessage(LowerCase(Trim(S)));          // تحويل لأحرف صغيرة
  ShowMessage(Copy(Trim(S), 1, 5));         // Hello
  ShowMessage(IntToStr(Pos('Delphi', S)));  // موضع الكلمة
  ShowMessage(StringReplace(S, 'World', 'Ring', []));
end;
🎯 المشروع: برنامج تحليل نص: عدد الكلمات، البحث عن كلمة، واستبدالها.

الخطوة 9: البرمجة الكائنية OOP

الكائنات تجمع البيانات (Fields) والسلوك (Methods) في وحدة واحدة. نتعلم: Class، Object، Constructor، Destructor، Inheritance، Encapsulation، Polymorphism.

type
  TPerson = class
  private
    FName: string;
    FAge: Integer;
  public
    constructor Create(const AName: string; AAge: Integer);
    procedure Introduce;
    property Name: string read FName write FName;
    property Age: Integer read FAge write FAge;
  end;

constructor TPerson.Create(const AName: string; AAge: Integer);
begin
  inherited Create;
  FName := AName;
  FAge := AAge;
end;

procedure TPerson.Introduce;
begin
  ShowMessage('أنا ' + FName + ' وعمري ' + IntToStr(FAge));
end;

// الاستخدام
var P: TPerson;
begin
  P := TPerson.Create('Ahmed', 25);
  try
    P.Introduce;
  finally
    P.Free;   // تحرير الذاكرة
  end;
end;
⚠️ كل كائن يُنشأ بـ Create يجب تحريره بـ Free داخل try..finally لتجنب تسريب الذاكرة.
🎯 المشروع: نظام لإدارة بيانات الأشخاص مع فئة مشتقة TEmployee ترث من TPerson.

الخطوة 16: التعامل مع JSON

JSON هي صيغة تبادل البيانات القياسية بين Delphi والويب. في Delphi نستخدم System.JSON وTJSONObject وTJSONArray.

{
  "id": 1,
  "name": "Ahmed",
  "age": 25,
  "skills": ["Delphi", "SQL", "JavaScript"]
}
uses System.JSON;

var
  Obj: TJSONObject;
  Arr: TJSONArray;
begin
  Obj := TJSONObject.Create;
  try
    Obj.AddPair('id', TJSONNumber.Create(1));
    Obj.AddPair('name', 'Ahmed');
    Obj.AddPair('age', TJSONNumber.Create(25));

    Arr := TJSONArray.Create;
    Arr.Add('Delphi');
    Arr.Add('SQL');
    Obj.AddPair('skills', Arr);

    Memo1.Lines.Text := Obj.Format;   // نص JSON منسق
  finally
    Obj.Free;
  end;
end;

الخطوة 17: التعامل مع الإنترنت و REST API

نتعلم أفعال HTTP (GET, POST, PUT, DELETE) ومفهوم REST، ونستخدم مكتبة REST Client في Delphi للاتصال بأي خدمة خارجية.

uses REST.Client, REST.Types;

var
  Client: TRESTClient;
  Request: TRESTRequest;
  Response: TRESTResponse;
begin
  Response := TRESTResponse.Create(nil);
  Client   := TRESTClient.Create('https://api.example.com');
  Request  := TRESTRequest.Create(nil);
  try
    Request.Client   := Client;
    Request.Response := Response;
    Request.Resource := 'customers';
    Request.Method   := rmGET;

    Request.Execute;

    if Response.StatusCode = 200 then
      Memo1.Lines.Text := Response.Content
    else
      ShowMessage('خطأ: ' + IntToStr(Response.StatusCode));
  finally
    Request.Free;
    Client.Free;
    Response.Free;
  end;
end;
🎯 المشروع: برنامج يقرأ بيانات من API عامة (مثل بيانات طقس) ويعرضها.

الخطوة 18: إنشاء تطبيق متعدد الطبقات

تقسيم المشروع إلى طبقات يجعل الصيانة والتوسع أسهل بكثير:

Presentation Layer → الشاشات والنماذج (UI) Business Logic Layer → القواعد والحسابات والتحقق Data Access Layer → الاستعلامات والاتصال بقاعدة البيانات Database Layer → الجداول والعلاقات
  • كل طبقة تتواصل مع الطبقة التي تليها فقط.
  • الواجهة لا تعرف SQL مباشرة أبداً.
  • تسهل إعادة استخدام المنطق في تطبيق ويب لاحقاً.

الخطوة 19: الأمان والحماية

  • التحقق من صحة المدخلات قبل الحفظ.
  • حماية كلمات المرور باستخدام Hash + Salt.
  • إدارة صلاحيات المستخدمين (Roles).
  • إدارة الجلسات وانتهاء صلاحيتها.
  • معالجة الأخطاء دون كشف تفاصيل داخلية.
  • تشفير الاتصال عبر HTTPS.
uses System.Hash;

function HashPassword(const Password, Salt: string): string;
begin
  Result := THashSHA2.GetHashString(Salt + Password);
end;

// عند تسجيل الدخول نقارن الهاش فقط، ولا نخزّن كلمة المرور أبداً
if HashPassword(InputPass, UserSalt) = StoredHash then
  ShowMessage('مرحباً بك')
else
  ShowMessage('بيانات الدخول غير صحيحة');
🔒 لا تخزّن كلمات المرور كنص صريح أبداً، ولا تبنِ استعلامات SQL بدمج النصوص مباشرة (استخدم Parameterized Queries لمنع SQL Injection).

CSS — الخطوة 6: التصميم المتجاوب

.container { padding: 12px; }
.sidebar  { display: none; }

@media (min-width: 768px) {
  .container { padding: 20px; }
}

/* كمبيوتر */
@media (min-width: 1024px) {
  .container { max-width: 1200px; margin: auto; }
  .sidebar   { display: block; }
}

JavaScript — الخطوة 9: التعامل مع البيانات

// قراءة البيانات المحفوظة أو مصفوفة فارغة
let tasks = JSON.parse(localStorage.getItem('tasks') || '[]');

function addTask(title) {
  tasks.push({ id: Date.now(), title, done: false });
  save();
}

function toggleTask(id) {
  const task = tasks.find(t => t.id === id);
  if (task) task.done = !task.done;
  save();
}

function save() {
  localStorage.setItem('tasks', JSON.stringify(tasks));
}

addTask('دراسة Delphi');
save();

المرحلة الثالثة — ربط Delphi بالويب

هذه أهم مرحلة: Delphi يعالج البيانات ويتصل بقاعدة البيانات، بينما HTML/CSS/JS تعرض البيانات وتتفاعل مع المستخدم، والوسيط بينهما هو REST API بصيغة JSON.

المستخدم | HTML + CSS + Bootstrap ← (العرض) | JavaScript ← (الطلب عبر fetch) | REST API ← (JSON) | Delphi ← (المنطق) | Database

أولاً: خدمة REST في Delphi

نجهّز مساراً باسم GET /api/customers يعيد قائمة العملاء بصيغة JSON:

procedure TCustomerResource.GetCustomers(Request: TWebRequest;
  Response: TWebResponse);
var
  Q: TFDQuery;
  Arr: TJSONArray;
  Obj: TJSONObject;
begin
  Arr := TJSONArray.Create;
  Q := TFDQuery.Create(nil);
  try
    Q.Connection := DM.FDConnection1;
    Q.SQL.Text := 'SELECT id, name, phone FROM Customers ORDER BY id';
    Q.Open;

    while not Q.Eof do
    begin
      Obj := TJSONObject.Create;
      Obj.AddPair('id',    TJSONNumber.Create(Q.FieldByName('id').AsInteger));
      Obj.AddPair('name',  Q.FieldByName('name').AsString);
      Obj.AddPair('phone', Q.FieldByName('phone').AsString);
      Arr.AddElement(Obj);
      Q.Next;
    end;

    Response.ContentType := 'application/json; charset=utf-8';
    Response.Content     := Arr.ToString;
  finally
    Q.Free;
    Arr.Free;
  end;
end;

ثالثاً: JavaScript يطلب البيانات

async function loadCustomers() {
  const tbody = document.getElementById('customers');

  try {
    const response = await fetch('/api/customers');
    if (!response.ok) throw new Error('HTTP ' + response.status);

    const customers = await response.json();

    tbody.innerHTML = customers.map(c => `
      <tr>
        <td>${escapeHtml(String(c.id))}</td>
        <td>${escapeHtml(c.name)}</td>
        <td>${escapeHtml(c.phone)}</td>
      </tr>
    `).join('');

  } catch (err) {
    tbody.innerHTML =
      '<tr><td colspan="3" class="text-danger text-center">تعذر تحميل البيانات</td></tr>';
    console.error(err);
  }
}

// حماية من حقن HTML
function escapeHtml(str) {
  return str.replace(/[&<>"']/g, ch => ({
    '&': '&amp;', '<': '<', '>': '>',
    '"': '"', "'": '''
  }[ch]));
}

loadCustomers();

ملاحظات أمنية مهمة

  • تحقق دائماً من صحة البيانات على الخادم (Delphi) وليس على المتصفح فقط.
  • استخدم Parameterized Queries لمنع SQL Injection.
  • استخدم escapeHtml عند عرض نصوص قادمة من المستخدم لمنع XSS.
  • أضف مصادقة (Token / JWT) لكل مسار API.
  • تعامل مع أخطاء الاتصال وأعد رسائل خطأ عامة للمستخدم.
  • استخدم HTTPS دائماً في الإنتاج.
